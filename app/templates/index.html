import requests

url = "http://localhost:8000/api/connectors/api/score"
headers = {
    "Content-Type": "application/json",
    "Authorization": "Bearer ..."
}
payload = {
    "workspace_id": "...",
    "data": [
        {
            "lead_id": "LEAD-101",
            "company_name": "Vertex Dynamics",
            "contact_person": "Sarah Connor",
            "lead_source": "Website",
            "engagement_level": "High",
            "interest_level": "High",
            "requirement_fit": "High",
            "days_since_interaction": 2,
            "deal_value_usd": 45000
        }
    ]
}

response = requests.post(url, json=payload, headers=headers)
enriched_leads = response.json()["results"]
for lead in enriched_leads:
    print(f"{lead['company_name']} -> Prob: {lead['conversion_probability_pct']}, Priority: {lead['priority']}")
